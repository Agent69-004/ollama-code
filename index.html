"☀️" : "🌙"; }); // Habit data let habits = JSON.parse(localStorage.getItem("habits")) || []; // Render habits function renderHabits() { const list = document.getElementById("habitsList"); list.innerHTML = ""; habits.forEach((habit, index) => { const li = document.createElement("li"); li.className = "habit-item"; li.innerHTML = `
    ${habit.name}
    `; list.appendChild(li); }); } // Add habit document.getElementById("habitForm").addEventListener("submit", (e) => { e.preventDefault(); const name = document.getElementById("habitName").value; const frequency = document.getElementById("frequency").value; const remind = document.getElementById("remindMe").checked; const streak = document.getElementById("streak").checked; habits.push({ name, frequency, remind, streak, completed: 0 }); localStorage.setItem("habits", JSON.stringify(habits)); renderHabits(); updateStats(); renderCalendar(); document.getElementById("habitForm").reset(); }); // Delete habit function deleteHabit(index) { habits.splice(index, 1); localStorage.setItem("habits", JSON.stringify(habits)); renderHabits(); updateStats(); renderCalendar(); } // Edit habit function editHabit(index) { const habit = habits[index]; const name = prompt("Edit habit name:", habit.name); if (name) { habits[index].name = name; localStorage.setItem("habits", JSON.stringify(habits)); renderHabits(); } } // Stats function updateStats() { const total = habits.length; const completed = habits.reduce((sum, h) => sum + h.completed, 0); document.getElementById("totalHabits").textContent = total; document.getElementById("completedHabits").textContent = completed; document.getElementById("totalHabitsSpan").textContent = total; } // Calendar function renderCalendar() { const calendar = document.getElementById("calendar"); calendar.innerHTML = ""; const today = new Date(); const month = today.getMonth(); const year = today.getFullYear(); const firstDay = new Date(year, month, 1); const lastDay = new Date(year, month + 1, 0); const daysInMonth = lastDay.getDate(); for (let i = 0; i < daysInMonth; i++) { const day = new Date(year, month, i + 1); const dayEl = document.createElement("div"); dayEl.className = "calendar-day"; dayEl.textContent = day.getDate(); calendar.appendChild(dayEl); } } // Chart function updateChart() { const ctx = document.getElementById("progressChart").getContext("2d"); const labels = ["Daily", "Weekly", "Monthly"]; const data = [0, 0, 0]; habits.forEach((habit) => { if (habit.frequency === "daily") data[0]++; else if (habit.frequency === "weekly") data[1]++; else if (habit.frequency === "monthly") data[2]++; }); new Chart(ctx, { type: "bar", data: { labels, datasets: [{ label: "Habits", data, backgroundColor: "rgba(52, 152, 219, 0.6)" }] }, options: { responsive: true, plugins: { legend: { display: false } } } }); } // Initialize renderHabits(); updateStats(); updateChart(); renderCalendar(); ```
